const c = require("crypto");
class Block {
    constructor(i, t, n1, n2, ph = "") {
        this.i = i;
        this.t = t;
        this.n1 = n1;
        this.n2 = n2;
        this.ph = ph;
        this.sum = n1 + n2;
        this.h = this.callHash();
    }
    callHash() {
        return c
            .createHash("sha256")
            .update(this.i + this.t + this.sum + this.ph)
            .digest("hex");
    }
}
class Blockchain {
    constructor() {
        this.chain = [this.createGBlock()];
    }
    createGBlock() {
        return new Block(0, Date.now(), 0, 0, "");
    }
    getCBlock() {
        return this.chain[this.chain.length - 1];
    }
    addBlock(b) {
        b.ph = this.getCBlock().h;
        b.h = b.callHash();
        this.chain.push(b);
    }
}
module.exports.Block = Block;

            .createHash("sha256")
            .update(this.i + this.t + this.km + this.ph)

            temp = parseInt(temp / 10);

        return sum == n;
    }
}

const c = require("crypto");
class Block {
    constructor(i, t, num, ph = "") {
        this.i = i;
        this.t = t;
        this.num = num;
        this.ph = ph;
        this.armStrong = this.checkArmStrong();
        this.h = this.callHash();
    }
    callHash() {
        return c
            .createHash("sha256")
            .update(this.i + this.t + this.num + this.armStrong + this.ph)
            .digest("hex");
    }
    checkArmStrong() {
        let n = this.num;
        if (!Number.isSafeInteger(n) || n < 0) return false;
        let counter = 0;
        let armStrongLength = String(n).length;
        while (n != 0) {
            let re = n % 10;
            counter = counter + Math.pow(re, armStrongLength);
            n = Math.floor(n / 10);
        }
        n = counter;
        if (n === this.num) return true;
        else return false;
    }
}

        this.chain = [this.createGBlock()];
    }

            this.chain.push(b);

    createGBlock() {
        return new Block(0, Date.now(), 3, "");
    }
    getCBlock() {
        return this.chain[this.chain.length - 1];

    validateBlock() {
        for (let i = 1; i < this.chain.length; i++) {

    constructor(i, t, n, ph = "") {
        this.i = i;
        this.t = t;
        this.ph = ph;
        this.n = n;
        this.H = this.checkHappyNumber(n);
        this.h = this.calHash();
    }
    calHash() {
        return c
            .createHash("sha256")
            .update(this.i + this.t + this.H + this.ph)
            .digest("hex");
    }
    checkHappyNumber(n) {
        if (!Number.isSafeInteger(n) || n <= 0) return "Not a Happy Number";
        let seen = new Set();
        while (n !== 1 && !seen.has(n)) {
            seen.add(n);
            n = this.sumOfSquares(n);
        }
        if (n === 1) {
            return "Happy Number";
        } else {
            return "Not a Happy Number";
        }
    }
    sumOfSquares(n) {
        let sum = 0;
        while (n > 0) {
            let digit = n % 10;
            sum += digit * digit;
            n = Math.floor(n / 10);
        }
        return sum;
    }
}

const c = require("crypto");
class Block {
    constructor(i, t, n, ph = "") {
        this.i = i;
        this.t = t;
        this.n = n;
        this.ph = ph;
        this.kaprekar = this.checkKaprekar();
        this.h = this.callHash();
    }
    callHash() {
        return c
            .createHash("sha256")
            .update(this.i + this.t + this.n + this.ph + this.kaprekar)
            .digest("hex");
    }
    checkKaprekar(n) {
        let num = this.n;
        if (!Number.isSafeInteger(num)) return false;
        if (num === 1) return true;
        if (num <= 0) return false;
        const square = (num * num).toString();
        const len = square.length;
        for (let i = 1; i < len; i++) {
            const left = parseInt(square.slice(0, i)) || 0; // left
            const right = parseInt(square.slice(i)) || 0; // right
            if (right > 0 && left + right === num) {
                return true;
            }
        }
        return false;
    }
}

import datetime


class Block:

    def __init__(self, i, op, bal, ph):
        self.i = i
        self.t = str(datetime.datetime.now())
        self.op = op
        self.bal = bal
        self.ph = ph
        self.h = self.calHash()

    def calHash(self):
        data = (str(self.i) + self.t + self.op + str(self.bal) + self.ph)
        return hashlib.sha256(data.encode()).hexdigest()


class Blockchain:

    def __init__(self):
        self.chain = []
        self.createGenesisBlock()

    def createGenesisBlock(self):
        genesis = Block(0, "d", 1000, "0")
        self.chain.append(genesis)

    def addBlock(self, op, m):
        pb = self.chain[-1].bal
        op = op.lower()
        if op not in ("d", "w") or m < 0:
            raise ValueError("Invalid transaction")
        if op == "d":
            pb += m
        elif op == "w":
            if pb < m:
                print("Error: The withdrawn balance is greater than the money.")
                return
            pb -= m
        ph = self.chain[-1].h
        b = Block(len(self.chain), op, pb, ph)
        self.chain.append(b)

    def display(self):
        for b in self.chain:
            print("=" * 50)
            print("Block:", b.i)
            print("Timestamp:", b.t)
            print("Balance:", b.bal)
            print("Operation:", b.op)
            print("Hash:", b.h)
            print("Previous Hash:", b.ph)

    def validate(self):
        for i in range(1, len(self.chain)):
            if self.chain[i].ph != self.chain[i-1].h:
                return False
            if self.chain[i].h != self.chain[i].calHash():
                return False
        return True

pragma solidity ^0.8.6;
contract Palindrone {
    function IsPalindrone(int num) public pure returns (bool) {
        if (num < 0) return false;
        if (num < 10) {
            return true;
        }
        int temporary = num;
        int reversed = 0;
        while (num > 0) {
            int remainder = num % 10;
            reversed = (reversed * 10) + remainder;
            num = num / 10;
        }
        return temporary == reversed;
    }
}

Demonstrate a Smart Contract to find mean, median, mode from a given array of numbers.

FILENAME: Stats.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;
contract Statistics {
    int[] public numbers;
    function AddNumber(int num) public {
        numbers.push(num);
    }
    // Calculating the Mean
    function mean() public view returns (int) {
        require(numbers.length > 0, "Array is empty");
        int sum = 0;
        for (uint i = 0; i < numbers.length; i++) {
            sum += numbers[i];
        }
        return sum / int(numbers.length);
    }
    // Calculating the Median
    function median() public view returns (int) {
        require(numbers.length > 0, "Array is Empty!");
        int[] memory arr = new int[](numbers.length);
        // Copying the original array into a temporary one
        for (uint i = 0; i < numbers.length; i++) {
            arr[i] = numbers[i];
        }
        // Sorting the new array
        for (uint i = 0; i < arr.length; i++) {
            for (uint j = i + 1; j < arr.length; j++) {
                if (arr[i] > arr[j]) {
                    int temp = arr[i];
                    arr[i] = arr[j];
                    arr[j] = temp;
                }
            }
        }
        uint n = arr.length;
        if (n % 2 == 0) {
            return (arr[n / 2 - 1] + arr[n / 2]) / 2;
        } else {
            return arr[n / 2];
        }
    }
    // Calculating the Mode
    function mode() public view returns (int) {
        require(numbers.length > 0, "Array is empty");
        int ModeValue = numbers[0];
        uint MaxCount = 1;
        for (uint i = 0; i < numbers.length; i++) {
            uint count = 1;
            for (uint j = i + 1; j < numbers.length; j++) {
                if (numbers[i] == numbers[j]) {
                    count++;
                }
            }
            if (count > MaxCount) {
                MaxCount = count;
                ModeValue = numbers[i];
            }
        }
        return ModeValue;
    }
}

contract perfectNumber {

    function isPerfect(uint num) public pure returns (string memory) {
        if (num <= 1) {
            return "Please enter values above one!";
        }
        uint sum = 1;
        // Locate all valid divisors
        for (uint i = 2; i <= num / 2; i++) {
            if (num % i == 0) {
                sum += i;
            }
        }
        if (sum == num) {
            return "The provided integer is Perfect.";
        } else {
            return "The provided integer is NOT Perfect.";
        }
    }
}

pragma solidity ^0.8.7;
contract voting {
    mapping(string => uint) public voteCount; //Stores the number of votes each candidate has
    mapping(address => bool) public voters; // Keeps a track of whather a candidate (address) has already been voted {true or false}
    string[] public candidateName; //Stores the list of all the candidate names
    constructor(string[] memory cdn) {
        candidateName = cdn; // Initializing with candidate names
    }
    function vote(string memory c) public {
        require(!voters[msg.sender], "You have already voted!"); // Checking if the voter have already voted prior
        bool candidateExists = false; // Variable to check whether the candidate exist's or not
        for (uint i = 0; i < candidateName.length; i++) {
            //Looping through all the candidates
            if (keccak256(bytes(c)) == keccak256(bytes(candidateName[i]))) {
                // Comparing using the Hash Value's instead of Strings
                candidateExists = true;
                break;
            }
        }
        require(candidateExists, "Candidate does not exist!"); // Checking if the Candidate Exists in the System
        voteCount[c]++;
        voters[msg.sender] = true;
    }
    function getWinner() public view returns (string memory) {
        string memory winner; //Stores the current winner
        uint highestVoteCount = 0;
        for (uint i = 0; i < candidateName.length; i++) {
            // Checking every Candidate
            if (voteCount[candidateName[i]] > highestVoteCount) {
                highestVoteCount = voteCount[candidateName[i]];
                winner = candidateName[i];
            }
        }
        return winner;
    }
}

        balance[msg.sender] += msg.value;
    }
    // Withdrawing Ether
    function withdraw(uint amount) public {
        require(balance[msg.sender] >= amount, "Insufficient Balance");
        balance[msg.sender] -= amount;
        payable(msg.sender).transfer(amount);
    }
    // Transferring Ether to another user
    function transfer(address payable receiver, uint amount) public {
        require(balance[msg.sender] >= amount, "Insufficient Balance");
        balance[msg.sender] -= amount;
        balance[receiver] += amount;
    }
    // Showing Account Balance
    function showBalance() public view returns (uint) {
        return balance[msg.sender];
    }
}

    }

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.21;
contract Vehicle {
    string vehicleRegistration;
    string vehicleBrand;
    uint vehicleRent;
    constructor(string memory regNo, string memory brand, uint price) {
        vehicleRegistration = regNo;
        vehicleBrand = brand;
        vehicleRent = price;
    }
}
contract CarRental is Vehicle {
    address renter;
    bool available = true;
    uint dueTime;
    event VehicleRented(address indexed renter, uint rentalDays, uint amount);
    event VehicleReturned(address indexed renter, uint penalty);
    constructor(string memory _regNo, string memory _brand, uint _price) Vehicle(_regNo, _brand, _price) {}
    function rentVehicle(uint _days) public payable {
        require(available, "Vehicle not available");
        require(_days > 0 && msg.value >= vehicleRent * _days, "Invalid days or payment");
        (available, renter, dueTime) = (false, msg.sender, block.timestamp + _days * 1 minutes);
        emit VehicleRented(msg.sender, _days, msg.value);
    }
    // 50% of daily price per late day (partial day counts as full)
    function calculatePenalty() public view returns (uint) {
        if (available || block.timestamp <= dueTime) return 0;
        return (((block.timestamp - dueTime) / 1 minutes + 1) * vehicleRent) / 2;
    }
    function returnVehicle() public payable {
        require(!available && msg.sender == renter, "Not renter or not rented");
        uint penalty = calculatePenalty();
        require(msg.value >= penalty, "Pay late penalty");
        available = true;
        emit VehicleReturned(msg.sender, penalty);
    }
}
